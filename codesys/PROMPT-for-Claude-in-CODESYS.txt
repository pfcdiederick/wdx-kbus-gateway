CONTEXT
-------
- Controller: WAGO PFC300 750-8302. The KBUS is NOT in the device tree. A program
  named kbus-gateway owns the KBUS (8 DI + 8 DO) and the serial port COM1, and
  publishes them as a WAGO Device Access (WDA) REST-API on
  http://127.0.0.1:8000/wda/parameters (JSON:API, HTTP/1.0, one request per connection).
- The project already talks to it: PLC_PRG calls FB_WdxKbusIo (the older function
  block) and drives a running light on DO1..DO8. That stays as it is.
- New, to add:
  * FB_WdaKbusIo - cyclic KBUS I/O every 5 ms using only standard WDA requests;
    reads the channel counts from the gateway; output xLocalControl when the
    dashboard has taken local control.
  * FB_WdaSerial - COM1 through the gateway: send with a rising edge on xSend,
    receive frames (xNewFrame, sFrame / abFrame); the COM1 settings (baud rate,
    format, frame gap) can be set with udiSetBaudrate / sSetFormat / uiSetFrameGap
    (0 / '' = keep) and are read in udiBaudrate / sFormat / uiFrameGap.
  * helper functions F_Wda*, and two example programs.

    i := i + 1;

=== OBJECT 3/14: F_FindBytes  (POU, Function, return type DINT, language ST) ===
--- DECLARATION ---
// ===========================================================================
// F_FindBytes  -  find a text in a byte buffer
// Object type in CODESYS: POU, Function, return type DINT, language ST
//
// Returns the 0-based position of sPattern in pBuf[udiFrom..udiLen-1], or -1.
// Works on buffers longer than 255 characters (Standard FIND does not).
// ===========================================================================
FUNCTION F_FindBytes : DINT
VAR_INPUT
    pBuf     : POINTER TO BYTE;
    udiLen   : UDINT;               // bytes in the buffer
    udiFrom  : UDINT;               // start position (0-based)
    sPattern : STRING(80);
END_VAR
VAR
    pP       : POINTER TO BYTE;
    udiPLen  : UDINT;
    i        : UDINT;
    j        : UDINT;
    xMatch   : BOOL;
END_VAR
--- IMPLEMENTATION ---
-----------------------------------------------------
F_FindBytes := -1;
udiPLen := TO_UDINT(LEN(sPattern));
pP := ADR(sPattern);
IF (udiPLen = 0) OR (udiLen < udiPLen) THEN
    RETURN;
END_IF
i := udiFrom;
WHILE i + udiPLen <= udiLen DO
    xMatch := TRUE;
    j := 0;
    WHILE xMatch AND (j < udiPLen) DO
        IF pBuf[i + j] <> pP[j] THEN
            xMatch := FALSE;
        END_IF
        j := j + 1;
    END_WHILE
    IF xMatch THEN
        F_FindBytes := TO_DINT(i);
        RETURN;
    END_IF
    i := i + 1;
END_WHILE
=== END OF F_FindBytes ===

=== OBJECT 4/14: F_WdaValuePos  (POU, Function, return type DINT, language ST) ===
--- DECLARATION ---
// ===========================================================================
// F_WdaValuePos  -  position of the value of one parameter in a WDA answer
// Object type in CODESYS: POU, Function, return type DINT, language ST
// Uses: F_FindBytes
//
// A WDA parameter resource looks like
//   {"type":"parameters","id":"...","attributes":{"path":"KBus/AnalogInputValues",
//    "dataType":"uint16","dataRank":"array","value":[0,16384],"stringValue":[...]}}
// Returns the 0-based position of the first character after "value": of the
// resource with this path, or -1 when the path is not in the answer.
// The closing quote is part of the search, so "KBus/DigitalInputs" does not
// match "KBus/DigitalInputs/1/Value".
// ===========================================================================
FUNCTION F_WdaValuePos : DINT
VAR_INPUT
    pBuf    : POINTER TO BYTE;
    udiLen  : UDINT;
    sPath   : STRING(60);           // e.g. 'KBus/DigitalInputValues'
END_VAR
VAR
    diPos   : DINT;
END_VAR
--- IMPLEMENTATION ---
-----------------------------------------------------
F_WdaValuePos := -1;
diPos := F_FindBytes(pBuf, udiLen, 0, CONCAT(CONCAT('"path":"', sPath), '"'));
IF diPos >= 0 THEN
    diPos := F_FindBytes(pBuf, udiLen, TO_UDINT(diPos), '"value":');
    IF diPos >= 0 THEN
        F_WdaValuePos := diPos + 8;
    END_IF
END_IF
=== END OF F_WdaValuePos ===

=== OBJECT 5/14: F_WdaCountItems  (POU, Function, return type INT, language ST) ===
--- DECLARATION ---
// ===========================================================================
// F_WdaCountItems  -  number of items in the array value of one parameter
// Object type in CODESYS: POU, Function, return type INT, language ST
// Uses: F_WdaValuePos, F_FindBytes
//
// For parameters of data type instantiations (KBus/DigitalInputs, ...): the
// WDA manual says only the NUMBER of items counts, their content is internal.
// Items are counted at the top level, so nested objects or strings are fine.
// Returns -1 when the path is not in the answer or the value is no array.
// ===========================================================================
FUNCTION F_WdaCountItems : INT
VAR_INPUT
    pBuf    : POINTER TO BYTE;
    udiLen  : UDINT;
    sPath   : STRING(60);
END_VAR
VAR
    diPos   : DINT;
    i       : UDINT;
    iDepth  : INT;
    iCommas : INT;
    xAny    : BOOL;
    c       : BYTE;
END_VAR
--- IMPLEMENTATION ---
-----------------------------------------------------
F_WdaCountItems := -1;
diPos := F_WdaValuePos(pBuf, udiLen, sPath);
IF diPos < 0 THEN
    RETURN;
END_IF
i := TO_UDINT(diPos);
WHILE (i < udiLen) AND (pBuf[i] = 32) DO               // spaces
    i := i + 1;
END_WHILE
IF (i >= udiLen) OR (pBuf[i] <> 91) THEN               // '['
    RETURN;
END_IF
i := i + 1;
WHILE i < udiLen DO
    c := pBuf[i];
    IF c = 34 THEN                                     // '"': skip the string
        xAny := TRUE;
        i := i + 1;
        WHILE (i < udiLen) AND (pBuf[i] <> 34) DO
            IF pBuf[i] = 92 THEN i := i + 1; END_IF     // '\' escapes the next byte
            i := i + 1;
        END_WHILE
    ELSIF (c = 91) OR (c = 123) THEN                   // '[' '{'
        xAny := TRUE;
        iDepth := iDepth + 1;
    ELSIF (c = 93) OR (c = 125) THEN                   // ']' '}'
        IF iDepth = 0 THEN
            EXIT;
        END_IF
        iDepth := iDepth - 1;
    ELSIF (c = 44) AND (iDepth = 0) THEN               // ','
        iCommas := iCommas + 1;
    ELSIF c > 32 THEN
        xAny := TRUE;
    END_IF
    i := i + 1;
END_WHILE
IF xAny THEN
    F_WdaCountItems := iCommas + 1;
ELSE
    F_WdaCountItems := 0;
END_IF
=== END OF F_WdaCountItems ===

=== OBJECT 9/14: F_WdaDecodeBytes  (POU, Function, return type INT, language ST) ===
--- DECLARATION ---
// ===========================================================================
// F_WdaDecodeBytes  -  JSON string value of one parameter -> bytes
// Object type in CODESYS: POU, Function, return type INT, language ST
// Uses: F_WdaValuePos, F_FindBytes
//
// For byte strings of the gateway (Serial/Com1/RxFrame, RxData, Format): one
// character per byte, printable ASCII as is, every other byte as \u00XX.
// "value":"AB\u000d\u000a" -> pDst[0..3] = 65, 66, 13, 10. At most uiMax bytes
// are stored. Returns the number of bytes, or -1 when the path is missing.
// ===========================================================================
FUNCTION F_WdaDecodeBytes : INT
VAR_INPUT
    pBuf    : POINTER TO BYTE;
    udiLen  : UDINT;
    sPath   : STRING(60);
    pDst    : POINTER TO BYTE;      // first element of the destination array
    uiMax   : UINT;                 // size of the destination array
END_VAR
VAR
    diPos   : DINT;
    i       : UDINT;
    j       : UDINT;
    uiN     : UINT;
    c       : BYTE;
    b       : BYTE;
    uiV     : UINT;
    h       : BYTE;
END_VAR
--- IMPLEMENTATION ---
-----------------------------------------------------
F_WdaDecodeBytes := -1;
diPos := F_WdaValuePos(pBuf, udiLen, sPath);
IF diPos < 0 THEN
    RETURN;
END_IF
i := TO_UDINT(diPos);
WHILE (i < udiLen) AND (pBuf[i] <> 34) DO              // up to the opening '"'
    i := i + 1;
END_WHILE
i := i + 1;
WHILE (i < udiLen) AND (pBuf[i] <> 34) DO              // up to the closing '"'
    c := pBuf[i];
    b := c;
    i := i + 1;
    IF (c = 92) AND (i < udiLen) THEN                  // '\' escape
        c := pBuf[i];
        i := i + 1;
        CASE c OF
        110: b := 10;                                  // \n
        114: b := 13;                                  // \r
        116: b := 9;                                   // \t
        98:  b := 8;                                   // \b
        102: b := 12;                                  // \f
        117:                                           // \uXXXX (gateway: <= 00FF)
            uiV := 0;
            FOR j := 0 TO 3 DO
                IF i < udiLen THEN
                    h := pBuf[i];
                    IF (h >= 48) AND (h <= 57) THEN uiV := uiV * 16 + (h - 48);
                    ELSIF (h >= 97) AND (h <= 102) THEN uiV := uiV * 16 + (h - 87);
                    ELSIF (h >= 65) AND (h <= 70) THEN uiV := uiV * 16 + (h - 55);
                    END_IF
                    i := i + 1;
                END_IF
            END_FOR
            b := UINT_TO_BYTE(uiV);
        ELSE
            b := c;                                    // \" \\ \/
        END_CASE
    END_IF
    IF uiN < uiMax THEN
        pDst[uiN] := b;
    END_IF
    uiN := uiN + 1;
END_WHILE
F_WdaDecodeBytes := TO_INT(MIN(uiN, uiMax));
=== END OF F_WdaDecodeBytes ===

=== OBJECT 10/14: FB_HttpTransaction  (POU, Function Block, language ST) ===
--- DECLARATION ---
// ===========================================================================
// FB_HttpTransaction  -  ONE HTTP request (byte based, fast completion)
// Object type in CODESYS: POU, Function Block, language ST
// Library: CAA Net Base Services (namespace NBS) + Standard
//
// Flow: connect -> send the complete request in ONE TCP write -> receive ->
// status code + body position. Speed:
//   - several stages run in the SAME task cycle where the library allows it
//     (connected -> send at once, sent -> read at once, complete -> parse)
//   - the answer is complete as soon as Content-Length bytes of body arrived
//     (the wdx interface always sends Content-Length); otherwise when the
//     server closes the connection, or 20 ms after the last data
//
// Gateway rules this FB respects:
//   - the gateway reads ONCE: the request must be one TCP write
//   - one client at a time: never run two transactions in parallel
//
// Usage: set pRequest/udiRequestLen, rising edge on xExecute, wait for xDone
//        or xError, then xExecute := FALSE (a FALSE->TRUE in the same cycle
//        starts the next request immediately).
// ===========================================================================
FUNCTION_BLOCK FB_HttpTransaction
VAR_INPUT
    xExecute      : BOOL;
    sHost         : STRING(15) := GVL_KbusRest.gc_sGatewayHost;
    uiPort        : UINT       := GVL_KbusRest.gc_uiGatewayPort;
    pRequest      : POINTER TO BYTE;
    udiRequestLen : UDINT;
    tTimeout      : TIME := T#1S;
END_VAR
VAR_OUTPUT
    xBusy         : BOOL;
    xDone         : BOOL;                                // valid HTTP answer (any status)
    xError        : BOOL;                                // no valid answer
    iStatus       : INT;                                 // 200, 204, 4xx; 0 = no answer
    pBody         : POINTER TO BYTE;
    udiBodyLen    : UDINT;
    sErrorText    : STRING(80);
END_VAR
VAR
    fbConn        : NBS.TCP_Connection;
    fbWrite       : NBS.TCP_Write;
    fbRead        : NBS.TCP_Read;
    ipAddr        : NBS.IP_ADDR;
    rtExecute     : R_TRIG;
    fbTimeout     : TON;
    fbIdle        : TON;
    iState        : INT;
    iPrev         : INT;
    iLoop         : INT;
    abChunk       : ARRAY[0..1023] OF BYTE;
    abRx          : ARRAY[0..8191] OF BYTE;              // 8 KB: a full WDA answer with stringValue and links
    udiRx         : UDINT;
    diHdrEnd      : DINT;                                // position of CR LF CR LF, -1 = not yet
    diCLen        : DINT;                                // Content-Length, -1 = none
    diPos         : DINT;
    sHead         : STRING(80);
    pHead         : POINTER TO BYTE;
    i             : UDINT;
    udiMax        : UDINT;
    xComplete     : BOOL;
END_VAR
--- IMPLEMENTATION ---
-----------------------------------------------------
rtExecute(CLK := xExecute);

// up to 4 stages per call: no task cycle is wasted between finished stages
FOR iLoop := 1 TO 4 DO
    iPrev := iState;
    CASE iState OF
    0:  // idle
        IF rtExecute.Q THEN
            xBusy := TRUE; xDone := FALSE; xError := FALSE;
            iStatus := 0; pBody := ADR(abRx); udiBodyLen := 0; sErrorText := '';
            udiRx := 0; diHdrEnd := -1; diCLen := -1;
            ipAddr.sAddr := sHost;
            IF (pRequest = 0) OR (udiRequestLen = 0) THEN
                sErrorText := 'no request';
                iState := 99;
            ELSE
                iState := 10;
            END_IF
        END_IF

    10: // connect
        fbConn(xEnable := TRUE, ipAddr := ipAddr, uiPort := uiPort);
        IF fbConn.xActive THEN
            iState := 20;
        ELSIF fbConn.xError THEN
            sErrorText := 'connect failed (gateway running? port?)';
            iState := 99;
        END_IF

    20: // send the whole request in ONE write
        fbConn(xEnable := TRUE, ipAddr := ipAddr, uiPort := uiPort);
        fbWrite(xExecute := TRUE, hConnection := fbConn.hConnection,
                szSize := udiRequestLen, pData := pRequest);
        IF fbWrite.xDone THEN
            fbWrite(xExecute := FALSE);
            iState := 30;
        ELSIF fbWrite.xError THEN
            fbWrite(xExecute := FALSE);
            sErrorText := 'send failed';
            iState := 99;
        END_IF

    30: // receive
        fbConn(xEnable := TRUE, ipAddr := ipAddr, uiPort := uiPort);
        fbRead(xEnable := TRUE, hConnection := fbConn.hConnection,
               szSize := SIZEOF(abChunk), pData := ADR(abChunk));
        IF fbRead.xReady AND fbRead.szCount > 0 THEN
            FOR i := 0 TO TO_UDINT(fbRead.szCount) - 1 DO   // szCount is CAA.SIZE (64 bit on ARM64)
                IF udiRx < SIZEOF(abRx) THEN
                    abRx[udiRx] := abChunk[i];
                    udiRx := udiRx + 1;
                END_IF
            END_FOR
            fbIdle(IN := FALSE);
            // header complete? then read Content-Length once
            IF diHdrEnd < 0 THEN
                diHdrEnd := F_FindBytes(ADR(abRx), udiRx, 0, '$R$N$R$N');
                IF diHdrEnd >= 0 THEN
                    diPos := F_FindBytes(ADR(abRx), TO_UDINT(diHdrEnd), 0, 'Content-Length: ');
                    IF diPos >= 0 THEN
                        diCLen := 0;
                        i := TO_UDINT(diPos) + 16;
                        WHILE (i < udiRx) AND (abRx[i] >= 48) AND (abRx[i] <= 57) DO
                            diCLen := diCLen * 10 + (abRx[i] - 48);
                            i := i + 1;
                        END_WHILE
                    END_IF
                END_IF
            END_IF
        END_IF
        xComplete := (diHdrEnd >= 0) AND (diCLen >= 0) AND (TO_DINT(udiRx) >= diHdrEnd + 4 + diCLen);
        fbIdle(IN := udiRx > 0, PT := T#20MS);
        IF xComplete OR ((udiRx > 0) AND (fbRead.xError OR NOT fbConn.xActive OR fbIdle.Q)) THEN
            iState := 40;
        ELSIF (udiRx = 0) AND (fbRead.xError OR NOT fbConn.xActive) THEN
            sErrorText := 'connection closed without answer';
            iState := 99;
        END_IF

    40: // parse status line and body position, close
        udiMax := MIN(udiRx, 80);
        pHead := ADR(sHead);
        FOR i := 0 TO udiMax - 1 DO
            pHead[i] := abRx[i];
        END_FOR
        pHead[udiMax] := 0;
        IF FIND(sHead, 'HTTP/1.') = 1 AND LEN(sHead) >= 12 THEN
            iStatus := STRING_TO_INT(MID(sHead, 3, 10));    // 'HTTP/1.0 200 OK' -> '200'
        END_IF
        IF diHdrEnd < 0 THEN
            diHdrEnd := F_FindBytes(ADR(abRx), udiRx, 0, '$R$N$R$N');
        END_IF
        IF diHdrEnd >= 0 THEN
            pBody := ADR(abRx[diHdrEnd + 4]);
            udiBodyLen := udiRx - TO_UDINT(diHdrEnd + 4);
        END_IF
        fbRead(xEnable := FALSE, hConnection := fbConn.hConnection);
        fbConn(xEnable := FALSE);
        fbIdle(IN := FALSE);
        xBusy := FALSE;
        IF iStatus > 0 THEN
            xDone := TRUE;
        ELSE
            xError := TRUE;
            sErrorText := 'no valid HTTP answer';
        END_IF
        iState := 0;

    99: // error: close everything
        fbWrite(xExecute := FALSE);
        fbRead(xEnable := FALSE, hConnection := fbConn.hConnection);
        fbConn(xEnable := FALSE);
        fbIdle(IN := FALSE);
        xBusy := FALSE;
        xError := TRUE;
        iState := 0;
    END_CASE

    // stop when nothing changed (waiting for the network) or finished
    IF (iState = iPrev) OR (iState = 0) THEN
        EXIT;
    END_IF
END_FOR

// overall time-out (also releases the gateway, which serves one client at a time)
fbTimeout(IN := (iState >= 10) AND (iState < 40), PT := tTimeout);
IF fbTimeout.Q THEN
    sErrorText := 'timeout';
    fbWrite(xExecute := FALSE);
    fbRead(xEnable := FALSE, hConnection := fbConn.hConnection);
    fbConn(xEnable := FALSE);
    xBusy := FALSE;
    xError := TRUE;
    iState := 0;
END_IF
=== END OF FB_HttpTransaction ===

// show what the exchange really needs.

    xReady := FALSE;
    xForceWrite := TRUE;

        ;                                                 // after an error: at most 10 tries per second

        diPos := F_FindBytes(p, udiN, 0, '"code":"');
        IF diPos >= 0 THEN
            j := TO_UDINT(diPos) + 8;

                iLastWdaCode := iLastWdaCode * 10 + (p[j] - 48);
                j := j + 1;
            END_WHILE
        END_IF
        diPos := F_FindBytes(p, udiN, 0, '"detail":"');   // first error: copy its detail
        IF diPos >= 0 THEN
            pErr := ADR(sLastError);
            j := 0;
            WHILE (TO_UDINT(diPos) + 10 + j < udiN) AND (j < 79) AND (p[TO_UDINT(diPos) + 10 + j] <> 34) DO
                pErr[j] := p[TO_UDINT(diPos) + 10 + j];
                j := j + 1;
            END_WHILE
            pErr[j] := 0;
        END_IF
    ELSE
        sLastError := fbHttp.sErrorText;
    END_IF
    xReady := FALSE;                                      // discover again (gateway restart, rescan)
    xForceWrite := TRUE;                                  // and send the outputs again
    iState := 0;
END_IF
END_FOR

END_IF
IF NOT xEnable THEN
    fbHttp(xExecute := FALSE);
    xOnline := FALSE;
    xFirst := TRUE;
    xKnown := FALSE;
    udiTriedBaud := 0; sTriedFormat := ''; uiTriedGap := 0;
    iState := 0;
    RETURN;
END_IF

// settings: which of the requested ones differ from the current ones (and were not tried yet)
xBaudDiff   := xKnown AND (udiSetBaudrate <> 0) AND (udiSetBaudrate <> udiBaudrate) AND (udiSetBaudrate <> udiTriedBaud);
xFormatDiff := xKnown AND (LEN(sSetFormat) = 3) AND (sSetFormat <> sFormat) AND (sSetFormat <> sTriedFormat);
xGapDiff    := xKnown AND (uiSetFrameGap <> 0) AND (uiSetFrameGap <> uiFrameGap) AND (uiSetFrameGap <> uiTriedGap);
xSettingsApplied := xKnown AND ((udiSetBaudrate = 0) OR (udiSetBaudrate = udiBaudrate))
                           AND ((LEN(sSetFormat) <> 3) OR (sSetFormat = sFormat))
                           AND ((uiSetFrameGap = 0) OR (uiSetFrameGap = uiFrameGap));

// ---------------------------------------------------------------- start a request
IF iState = 0 THEN
    IF (uiErrStreak > 0) AND (TIME() - tLastPoll < T#200MS) THEN
        ;                                                 // after an error: at most 5 tries per second
    ELSIF xBaudDiff OR xFormatDiff OR xGapDiff THEN
        // {"data":[{"type":"parameters","id":"0-0-serial-com1-baudrate","attributes":{"value":19200}}, ...]}
        udiBody := 0;
        xNeedComma := FALSE;
        F_AppendStr(ADR(abBody), SIZEOF(abBody), '{"data":[', udiBody);
        IF xBaudDiff THEN
            F_AppendStr(ADR(abBody), SIZEOF(abBody), '{"type":"parameters","id":"0-0-serial-com1-baudrate","attributes":{"value":', udiBody);
            F_AppendStr(ADR(abBody), SIZEOF(abBody), CONCAT(UDINT_TO_STRING(udiSetBaudrate), '}}'), udiBody);
            xNeedComma := TRUE;
        END_IF
        IF xFormatDiff THEN
            IF xNeedComma THEN F_AppendStr(ADR(abBody), SIZEOF(abBody), ',', udiBody); END_IF
            F_AppendStr(ADR(abBody), SIZEOF(abBody), '{"type":"parameters","id":"0-0-serial-com1-format","attributes":{"value":"', udiBody);
            F_AppendStr(ADR(abBody), SIZEOF(abBody), CONCAT(sSetFormat, '"}}'), udiBody);
            xNeedComma := TRUE;
        END_IF
        IF xGapDiff THEN
            IF xNeedComma THEN F_AppendStr(ADR(abBody), SIZEOF(abBody), ',', udiBody); END_IF
            F_AppendStr(ADR(abBody), SIZEOF(abBody), '{"type":"parameters","id":"0-0-serial-com1-framegap","attributes":{"value":', udiBody);
            F_AppendStr(ADR(abBody), SIZEOF(abBody), CONCAT(UINT_TO_STRING(uiSetFrameGap), '}}'), udiBody);
        END_IF
        F_AppendStr(ADR(abBody), SIZEOF(abBody), ']}', udiBody);
        udiTx := 0;
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'PATCH /wda/parameters HTTP/1.0$R$N', udiTx);
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'Content-Type: application/vnd.api+json$R$N', udiTx);
        F_AppendStr(ADR(abTx), SIZEOF(abTx), CONCAT(CONCAT('Content-Length: ', UDINT_TO_STRING(udiBody)), '$R$N$R$N'), udiTx);
        FOR j := 0 TO udiBody - 1 DO
            abTx[udiTx] := abBody[j]; udiTx := udiTx + 1;
        END_FOR
        udiTriedBaud := udiSetBaudrate;                   // also when rejected: no endless retries
        sTriedFormat := sSetFormat;
        uiTriedGap := uiSetFrameGap;
        iState := 3;
    ELSIF xSendPending THEN
        // body {"data":{"type":"parameters","id":"0-0-serial-com1-transmit","attributes":{"value":"..."}}}
        udiBody := 0;
        F_AppendStr(ADR(abBody), SIZEOF(abBody), '{"data":{"type":"parameters","id":"0-0-serial-com1-transmit",', udiBody);
        F_AppendStr(ADR(abBody), SIZEOF(abBody), '"attributes":{"value":"', udiBody);
        IF uiTxLen > 0 THEN
            uiLen := MIN(uiTxLen, 256);
            pS := ADR(abTxData);
        ELSE
            uiLen := TO_UINT(LEN(sTxData));
            pS := ADR(sTxData);
        END_IF
        pHex := ADR(sHex);
        FOR i := 0 TO uiLen - 1 DO                        // one character per byte: \u00XX outside printable ASCII
            IF uiLen = 0 THEN EXIT; END_IF
            b := pS[i];
            IF (b >= 32) AND (b < 127) AND (b <> 34) AND (b <> 92) THEN
                abBody[udiBody] := b; udiBody := udiBody + 1;
            ELSE
                F_AppendStr(ADR(abBody), SIZEOF(abBody), '\u00', udiBody);
                abBody[udiBody] := pHex[b / 16]; abBody[udiBody + 1] := pHex[b MOD 16];
                udiBody := udiBody + 2;
            END_IF
        END_FOR
        F_AppendStr(ADR(abBody), SIZEOF(abBody), '"}}}', udiBody);
        udiTx := 0;
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'PATCH /wda/parameters/0-0-serial-com1-transmit HTTP/1.0$R$N', udiTx);
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'Content-Type: application/vnd.api+json$R$N', udiTx);
        F_AppendStr(ADR(abTx), SIZEOF(abTx), CONCAT(CONCAT('Content-Length: ', UDINT_TO_STRING(udiBody)), '$R$N$R$N'), udiTx);
        FOR j := 0 TO udiBody - 1 DO
            abTx[udiTx] := abBody[j]; udiTx := udiTx + 1;
        END_FOR
        xSendPending := FALSE;
        xSendBusy := TRUE;
        iState := 1;
    ELSIF TIME() - tLastPoll >= tPoll THEN
        udiTx := 0;
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'GET /wda/parameters?filter[path]=Serial/Com1/Open,Serial/Com1/Baudrate,', udiTx);
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'Serial/Com1/Format,Serial/Com1/FrameGap,Serial/Com1/TxCount,', udiTx);
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'Serial/Com1/RxCount,Serial/Com1/RxFrame,Serial/Com1/RxFrameCount,', udiTx);
        F_AppendStr(ADR(abTx), SIZEOF(abTx), 'Serial/Com1/Errors HTTP/1.0$R$N$R$N', udiTx);
        iState := 2;
    END_IF
    IF iState > 0 THEN
        tLastPoll := TIME();
        fbHttp.pRequest := ADR(abTx);
        fbHttp.udiRequestLen := udiTx;
        fbHttp(xExecute := FALSE);                        // FALSE -> TRUE in the same cycle
    END_IF
END_IF

// ---------------------------------------------------------------- run request
IF iState > 0 THEN
    fbHttp(xExecute := TRUE, sHost := sHost, uiPort := uiPort, tTimeout := tTimeout);
    IF fbHttp.xDone OR fbHttp.xError THEN
        iLastStatus := fbHttp.iStatus;
        p := fbHttp.pBody;
        udiN := fbHttp.udiBodyLen;
        IF fbHttp.xDone AND ((((iState = 1) OR (iState = 3)) AND (iLastStatus = 204)) OR ((iState = 2) AND (iLastStatus = 200))) THEN
            IF iState = 3 THEN
                tLastPoll := TIME() - tPoll;              // settings written: read them back at once
            ELSIF iState = 1 THEN
                xSendBusy := FALSE;
                xSendDone := TRUE;
            ELSE
                xOpen := F_WdaParseScalar(p, udiN, 'Serial/Com1/Open') = 1;
                udiBaudrate := LINT_TO_UDINT(MAX(F_WdaParseScalar(p, udiN, 'Serial/Com1/Baudrate'), 0));
                uiFrameGap := LINT_TO_UINT(MAX(F_WdaParseScalar(p, udiN, 'Serial/Com1/FrameGap'), 0));
                udiTxBytes := LINT_TO_UDINT(MAX(F_WdaParseScalar(p, udiN, 'Serial/Com1/TxCount'), 0));
                udiRxBytes := LINT_TO_UDINT(MAX(F_WdaParseScalar(p, udiN, 'Serial/Com1/RxCount'), 0));
                udiLineErrors := LINT_TO_UDINT(MAX(F_WdaParseScalar(p, udiN, 'Serial/Com1/Errors'), 0));
                iN := F_WdaDecodeBytes(p, udiN, 'Serial/Com1/Format', ADR(sFormat), 3);
                IF iN >= 0 THEN pS := ADR(sFormat); pS[iN] := 0; END_IF
                liVal := F_WdaParseScalar(p, udiN, 'Serial/Com1/RxFrameCount');
                IF (liVal >= 0) AND (LINT_TO_UDINT(liVal) <> udiFrames) THEN
                    iN := F_WdaDecodeBytes(p, udiN, 'Serial/Com1/RxFrame', ADR(abFrame), 512);
                    uiFrameLen := TO_UINT(MAX(iN, 0));
                    pS := ADR(sFrame);                    // as text: up to 255 bytes
                    FOR i := 0 TO 254 DO
                        IF i >= uiFrameLen THEN EXIT; END_IF
                        pS[i] := abFrame[i];
                    END_FOR
                    pS[MIN(uiFrameLen, 255)] := 0;
                    xNewFrame := NOT xFirst;              // not for the frame that was there before the start
                    udiFrames := LINT_TO_UDINT(liVal);
                END_IF
                xFirst := FALSE;
                xKnown := TRUE;
            END_IF
            uiErrStreak := 0;
            xOnline := TRUE;
            iLastWdaCode := 0;
        ELSE
            // 400/500 with WDA code (e.g. 21 = COM1 not open, 39 = character > U+00FF),
            // 404 = gateway without COM1 (SERIAL_DEVICE empty or older than 2.5.0)
            uiErrStreak := uiErrStreak + 1;
            xOnline := uiErrStreak < 3;
            IF iState = 1 THEN xSendBusy := FALSE; END_IF
            iLastWdaCode := 0;
            IF fbHttp.xDone THEN
                sLastError := CONCAT('HTTP ', INT_TO_STRING(iLastStatus));
                diPos := F_FindBytes(p, udiN, 0, '"code":"');
                IF diPos >= 0 THEN
                    j := TO_UDINT(diPos) + 8;
                    WHILE (j < udiN) AND (p[j] >= 48) AND (p[j] <= 57) DO
                        iLastWdaCode := iLastWdaCode * 10 + (p[j] - 48);
                        j := j + 1;
                    END_WHILE
                END_IF
                diPos := F_FindBytes(p, udiN, 0, '"detail":"');
                IF diPos >= 0 THEN
                    pErr := ADR(sLastError);
                    j := 0;
                    WHILE (TO_UDINT(diPos) + 10 + j < udiN) AND (j < 79) AND (p[TO_UDINT(diPos) + 10 + j] <> 34) DO
                        pErr[j] := p[TO_UDINT(diPos) + 10 + j];
                        j := j + 1;
                    END_WHILE
                    pErr[j] := 0;
                END_IF
            ELSE
                sLastError := fbHttp.sErrorText;
            END_IF
        END_IF
        fbHttp(xExecute := FALSE);
        iState := 0;
    END_IF
END_IF
=== END OF FB_WdaSerial ===
